## 5. 隔离CPU见证的性质与可运行代码

实际在当前容器运行了下面的程序。它**不是直接导入安装好的完整fork**：`test_passed/test_failed`、两个parser保留已读函数的判断逻辑；报告循环和ratio为等价缩减转写，移除文件/容器依赖并改变报告结构。mypy正则来自固定源码；shell用合成命令。故证据级别是“源逻辑的隔离行为见证”，不是完整上游包回归测试或Docker任务复现。

程序没有网络调用、不改用户仓库、不使用真实gold答案。B可以先按原文件导入重跑，再使用真实任务日志补集成对照。

```python
"""Isolated CPU witnesses for SWE-Gym/SWE-Bench-Fork@242429c1.
Copied predicates/reduction semantics, with imports and reporting reduced.
Not a Docker run, installed-package test, or assessment of real task frequency.
"""
from enum import Enum
import json, re, subprocess

class TestStatus(Enum):
    FAILED = 'FAILED'
    PASSED = 'PASSED'
    SKIPPED = 'SKIPPED'
    ERROR = 'ERROR'
    XFAIL = 'XFAIL'

def test_passed(case, sm):
    return case in sm and sm[case] in [TestStatus.PASSED.value, TestStatus.XFAIL.value]

def test_failed(case, sm):
    return case not in sm or any(sm[case] == status for status in [TestStatus.FAILED.value, TestStatus.ERROR.value])

def counts(sm, tests):
    success, failure = [], []
    for case in tests:
        if test_passed(case, sm):
            success.append(case)
        elif test_failed(case, sm):
            failure.append(case)
    return success, failure

def ratio(count):
    s, f = count
    return len(s) / (len(s) + len(f)) if len(s) + len(f) else 1

def resolve(sm, f2p=('f',), p2p=('p',)):
    a,b = counts(sm,f2p),counts(sm,p2p)
    f,p=ratio(a),ratio(b)
    status = 'FULL' if f == 1 and p == 1 else 'PARTIAL' if 0 < f < 1 and p == 1 else 'NO'
    return {'status': status, 'f2p_count': len(a[0])+len(a[1]), 'p2p_count':len(b[0])+len(b[1])}

def parse_log_pytest(log):
    test_status_map = {}
    for line in log.split('\n'):
        if any(line.startswith(x.value) for x in TestStatus):
            if line.startswith(TestStatus.FAILED.value):
                line = line.replace(' - ', ' ')
            test_case = line.split()
            if len(test_case) <= 1:
                continue
            test_status_map[test_case[1]] = test_case[0]
    return test_status_map

def parse_log_pytest_pydantic(log):
    test_status_map = {}
    escapes = ''.join(chr(char) for char in range(1,32))
    for line in log.split('\n'):
        line = re.sub(r'\[(\d+)m', '', line)
        translator = str.maketrans('', '', escapes)
        line = line.translate(translator)
        line = re.sub(r'FAILED\s*\[.*?\]', 'FAILED', line)
        if 'tests/test_main.py::test_model_post_init_supertype_private_attr' in line:
            print(line)
        if any(line.startswith(x.value) for x in TestStatus):
            if line.startswith(TestStatus.FAILED.value):
                line = line.replace(' - ', ' ')
            test_case = line.split()
            test_status_map[test_case[1]] = test_case[0]
        elif any(line.endswith(x.value) for x in TestStatus):
            test_case = line.split()
            test_status_map[test_case[0]] = test_case[1]
    return test_status_map

out={}
for k,sm in [('passed',{'f':'PASSED','p':'PASSED'}),('missing',{}),('all_skipped',{'f':'SKIPPED','p':'SKIPPED'}),('all_xfail',{'f':'XFAIL','p':'XFAIL'}),('error_f',{'f':'ERROR','p':'PASSED'})]:
    out[k]=resolve(sm)
out['empty_references']=resolve({},(),())
out['empty_p2p']=resolve({'f':'PASSED'},('f',),())
out['mixed_skip_fail']=resolve({'f1':'PASSED','f2':'SKIPPED','p':'PASSED'},('f1','f2'),('p',))
# run_validation's intermediate report includes only literal PASSED and FAILED.
sm={'f':'PASSED','p':'ERROR'}
out['validation_pass_plus_error']=(len([v for v in sm.values() if v=='FAILED'])==0 and len([v for v in sm.values() if v=='PASSED'])>0)
out['generic_prefix_summary']=parse_log_pytest('PASSED tests/t.py::test_a\nFAILED tests/t.py::test_b - AssertionError')
out['generic_verbose_only']=parse_log_pytest('tests/t.py::test_a PASSED')
out['generic_color']=parse_log_pytest('\x1b[32mPASSED\x1b[0m tests/t.py::test_a')
out['pydantic_color']=parse_log_pytest_pydantic('\x1b[32mPASSED\x1b[0m tests/t.py::test_a')
out['test_id_space_truncation']=parse_log_pytest('PASSED tests/t.py::test_a[hello world]')
out['pydantic_bracket']=parse_log_pytest_pydantic('FAILED [gw0] tests/t.py::test_b - AssertionError')
try: parse_log_pytest_pydantic('PASSED')
except IndexError: out['pydantic_malformed_status']='IndexError'
# exact regex used by make_test_command for mypy
patch='- [case old_case]\n+ [case new_case]\n  [case context_case]'
out['mypy_selector']='pytest -rA -k ' + '"' + ' or '.join(re.findall(r'\[case ([^\]]+)\]',patch)) + '"'
# Demonstrate shell lifecycle independently, with no repository or Docker.
p=subprocess.run(['/bin/bash','-c','set -xo pipefail; false; true'],capture_output=True,text=True)
out['eval_shell_last_command_exit']=p.returncode
assert out['all_skipped']['status']=='FULL' and out['missing']['status']=='NO'
assert out['all_skipped']['f2p_count']==0
assert out['all_xfail']['status']=='FULL' and out['error_f']['status']=='NO'
assert out['empty_references']['status']=='FULL'
assert out['validation_pass_plus_error'] is True
assert out['generic_color']=={} and out['generic_verbose_only']=={}
assert out['pydantic_malformed_status']=='IndexError'
assert out['eval_shell_last_command_exit']==0
print(json.dumps(out,indent=2,ensure_ascii=False))
```
