    def test_load_ignore_warning(self, msg_box):
        self.widget._parse_var_defs(dict(categorical={}, numeric={}))
        msg_box.assert_not_called()

        def shown_text():
            msg_box.assert_called()
            # Permit either positional or keyword text and multiple warnings.
            return "\n".join(
                call[0][2] if len(call[0]) >= 3 else call[1].get("text", "")
                for call in msg_box.call_args_list)

        no_change = dict(renamed_values={}, colors={})
        all_names = ("foo", "bar", "baz", "qux", "quux", "corge", "grault")
        for n in range(1, len(all_names) + 1):
            names = all_names[:n]
            msg_box.reset_mock()
            self.widget._parse_var_defs(dict(
                categorical=dict.fromkeys(names, no_change),
                numeric={}))
            text = shown_text()
            named = [name for name in names if name in text]
            if n <= 2:
                self.assertEqual(named, list(names), text)
            else:
                # Long lists may be abbreviated. No punctuation, order,
                # truncation threshold or omitted-count wording is prescribed.
                self.assertTrue(named, text)

        # A mixed file must report unused definitions in either section while
        # continuing to apply definitions for variables present in the data.
        self.send_signal(self.widget.Inputs.data, self.iris)
        msg_box.reset_mock()
        self.widget._parse_var_defs(
            {"categorical": {"iris": {"rename": "species"},
                             "foo": {"renamed_values": {}}},
             "numeric": {"bar": {"colors": "linear_viridis"}}})
        text = shown_text()
        self.assertIn("foo", text)
        self.assertIn("bar", text)
        out = self.get_output(self.widget.Outputs.data)
        self.assertIsNotNone(out)
        self.assertEqual(out.domain.class_var.name, "species")
