        ref = context.recorder.write_json_artifact(             # 拒绝也落 audit
            "stage16g2b_file_mutation_denial", denial_payload, ...)
        return _tool_result(..., status="denied", typed={..., "denial_ref": ref...})

    # 成功 / partial_failure 路径
    for path in outcome.cache_removals: context.file_state_cache.pop(path, None)
    context.file_state_cache.update(outcome.cache_updates)         # 更新读缓存
    ref = context.recorder.write_json_artifact(                    # 落 audit
        "stage16g2b_file_mutation_audit", outcome.audit_payload(), ...)
    status = "error" if outcome.partial_failure else "ok"
    return _tool_result(
        ..., status=status,
        content=_preview(json.dumps({                              # 模型可见 JSON 预览
            "result_kind": outcome.result_kind,
            "operation_count": len(outcome.operation_facts),
            "changed_paths": outcome.changed_paths,
            "partial_failure": outcome.partial_failure,
            "rollback_status": outcome.rollback_status,
            "audit_artifact_id": ref.artifact_id,                # ⚠ 16G.2C 改名
        })),
        artifact_refs=[ref],
        typed={**outcome.audit_payload(), "audit_artifact": ref...,
               "allowed_in_policy_loss_trajectory": False, ...},
    )

④ 共享核心 execute_structured_file_mutation（file_mutation.py:177）

这是原子语义的关键代码段。30 行内说清"预检通过才动文件"和"任一失败 rollback 全部"：

def execute_structured_file_mutation(*, tool_name, arguments, context) -> FileMutationOutcome:
    operations = _normalize_operations(tool_name, arguments)        # ① 5 件套统一形态
    prepared   = _prepare_operations(operations, context)           # ② 预检全部 op
    snapshot_paths = _snapshot_paths(prepared, context)             # ③ 算出 affected 路径集
    snapshots = {path: _snapshot_path(path) for path in snapshot_paths}
    applied_ids = []
    try:
        for mutation in prepared:
            _apply_prepared_operation(mutation)                     # ④ 真正动文件
            applied_ids.append(mutation.operation_id)
    except Exception as exc:                                  # ⑤ rollback 全部
        rollback_status = _rollback_snapshots(snapshots)
        operation_facts = [m.fact() for m in prepared]
        operation_facts.append({"result_kind": "runtime_apply_failed",
                                 "applied_operation_ids": applied_ids,
                                 "failed_error_type": type(exc).__name__})
        return FileMutationOutcome(
            tool_name=tool_name,
            result_kind="partial_failure",
            operation_facts=operation_facts,
            ...
            partial_failure=True,
            rollback_attempted=True,
            rollback_status=rollback_status,
            invalid_for_training=True,                          # ⚠ 训练阻断
            policy_loss_candidate=False,
            diagnostic_side_channel_only=True,
        )
    # 成功路径：更新 cache，构造 outcome
    return FileMutationOutcome(
        tool_name=tool_name, result_kind="file_mutation_applied",
        operation_facts=[m.fact() for m in prepared],
        changed_paths=sorted(_relative_touched_paths(prepared)),
        ...
        policy_loss_candidate=False,                              # ⚠ 永远是 False
        official_prediction_eligible=False,
        training_export_eligible=False,
    )

⑤ _normalize_operations — 5 件套 → 统一 ops list（file_mutation.py:238-276）

覆盖阅读路径步骤 7。纯转换函数，不读盘、不调外部依赖。把 5 种 tool_name × 5 种 arguments 形态都展开成同一种数据形态 list[dict{op, path, ...}]，让 step 8 之后的代码只需要处理一种 ops 列表。

def _normalize_operations(tool_name: str, arguments: dict[str, Any]) -> list[dict[str, Any]]:
    if tool_name == "apply_patch":
        return [dict(operation) for operation in arguments["operations"]]    # 已是 list[dict]，直接复制
    if tool_name == "write_file":
        return [
            {
                "op": "write_file",
                "path": arguments["path"],
                "content": arguments["content"],
                "mode": arguments["mode"],
                **({"expected_content_hash": arguments["expected_content_hash"]}
                   if "expected_content_hash" in arguments else {}),
            }
        ]
    if tool_name == "delete_file":
        return [{"op": "delete_file", "path": arguments["path"],
                 "expected_content_hash": arguments["expected_content_hash"],
                 "reason": arguments["reason"]}]
    if tool_name == "move_file":
        return [{"op": "move_file",
                 "source_path": arguments["source_path"],
                 "target_path": arguments["target_path"],
                 "expected_source_hash": arguments["expected_source_hash"],
                 "reason": arguments["reason"]}]
    if tool_name == "mkdir":
        return [{"op": "mkdir", "path": arguments["path"]}]
    raise FileMutationDenial(
        reason_code="unsupported_file_mutation_tool",
        message=f"{tool_name} is not a Stage 16G.2B file mutation tool.",
        retryable=False,
    )

读懂这段：apply_patch 是天然多 op 形态，本函数原样透传；其他 4 个单 op 工具被人工包成 1 元 list 让形态统一。如果 schema 校验放过了某个未知 tool_name（不应该发生），最后兜底 raise unsupported_file_mutation_tool。delete/move 强制带 reason 字段（schema 必填）让这里直接 arguments["reason"] 不报 KeyError。

⑥ _prepare_operations 预检 — 写盘前的安全闸门（file_mutation.py:279-309 + 312-332）

覆盖阅读路径步骤 8。原子语义的核心保证：所有 op 的预检都走完，没有任何一个 raise，才进入步骤 9 真正写盘。任一失败 → raise FileMutationDenial 一路冒泡到 handler → 翻成 status="denied" ToolResult，磁盘文件一字不改。

⑥.1 主循环：限总写入量 + 拒绝重复路径

def _prepare_operations(operations: list[dict[str, Any]], context: Any) -> list[PreparedFileMutation]:
    total_write_bytes = 0
    prepared: list[PreparedFileMutation] = []
    touched: set[Path] = set()
    for index, operation in enumerate(operations):
        op_name = str(operation["op"])
        operation_id = f"op_{index:03d}"                                  # op_000 / op_001 ...
        mutation = _prepare_operation(operation_id, op_name, operation, context)  # 单 op 预检
        total_write_bytes += _operation_write_bytes(mutation)
        if total_write_bytes > STAGE16G2_APPLY_PATCH_TOTAL_WRITE_MAX_BYTES:
            raise FileMutationDenial(reason_code="patch_too_large", retryable=True,
                operation_id=operation_id,
                safe_rewrite_example="Split the change into smaller structured apply_patch calls.")
        for path in mutation.touched_paths:
            if path in touched:                                          # 同次调用重复触碰
                raise FileMutationDenial(reason_code="duplicate_mutation_path",
                    operation_id=operation_id,
                    safe_alternative_tool="write_file",
                    safe_rewrite_example="Use one write_file overwrite for a large same-file rewrite.")
            touched.add(path)
        prepared.append(mutation)
    return prepared

def _prepare_operation(operation_id, op_name, operation, context) -> PreparedFileMutation:
    if op_name == "replace_text":  return _prepare_replace_text(operation_id, operation, context)
    if op_name == "write_file":    return _prepare_write_file(operation_id, operation, context)
    if op_name == "delete_file":   return _prepare_delete_file(operation_id, operation, context)
    if op_name == "move_file":     return _prepare_move_file(operation_id, operation, context)
    if op_name == "mkdir":         return _prepare_mkdir(operation_id, operation, context)
    raise FileMutationDenial(reason_code="unsupported_apply_patch_operation",
        message=f"Unsupported apply_patch operation: {op_name}",
        operation_id=operation_id)

def _prepare_replace_text(operation_id, operation, context) -> PreparedFileMutation:
    path = _normalize_relative_path(operation["path"])              # L2 + L3 路径校验
    resolved = _resolve_workspace_path(context, path, must_exist=True)  # 拒绝符号链接 + 必须已存在
    content = _read_existing_utf8_file(resolved, path, operation_id=operation_id)  # UTF-8 拒绝 binary
    current_hash = _sha256_text(content)
    expected_hash = operation["expected_content_hash"]
    if expected_hash != current_hash:                              # 检测脏读
        raise FileMutationDenial(reason_code="stale_file_state", path=path, ...
            safe_rewrite_example=f"Call read_file(path={path!r}) and retry with the returned content_hash.")
    old_text = operation["old_text"]
    if _looks_like_numbered_read_file_snippet(old_text):            # 拒绝带 read_file 行号前缀
        raise FileMutationDenial(reason_code="line_number_prefix_in_old_text", ...
            safe_rewrite_example="Copy old_text from raw_content_preview, not numbered_content_preview.")
    count = content.count(old_text)
    if count != 1:                                                  # old_text 必须唯一
        raise FileMutationDenial(reason_code="old_text_not_unique", ...)
    # 通过所有校验，构造 PreparedFileMutation 并返回

读懂这段：这是"模型脏读 + 改 / 把行号前缀也复制进来 / 同一段文字在文件里出现多次"这三个最常见模型错误的拦截点。每个 reason_code 都附 safe_rewrite_example，handler 把它带进 model-visible 错误返回，模型 turn N+1 可以直接看懂怎么改。其他 4 个 _prepare_*（write_file / delete_file / move_file / mkdir）结构类似，本节略。

⑦ 真实文件操作 _apply_prepared_operation（file_mutation.py:555）

所有"真的写盘 / 删文件 / 重命名 / 建目录"都在这里。其他地方不能直接调 Path.unlink。

def _apply_prepared_operation(mutation: PreparedFileMutation) -> None:
    if mutation.op in {"replace_text", "write_file"}:
        mutation.resolved_path.parent.mkdir(parents=True, exist_ok=True)  # 自动建父目录
        mutation.resolved_path.write_text(mutation.content, encoding="utf-8")
        return
    if mutation.op == "delete_file":
        mutation.resolved_path.unlink()
        return
    if mutation.op == "move_file":
        mutation.resolved_target_path.parent.mkdir(parents=True, exist_ok=True)
        mutation.resolved_source_path.rename(mutation.resolved_target_path)
        return
    if mutation.op == "mkdir":
        mutation.resolved_path.mkdir(parents=True, exist_ok=True)
        return
    raise RuntimeError(f"unhandled file mutation op: {mutation.op}")

注意：parents=True, exist_ok=True 在 write_file / move_file 中自动建父目录 — 这是故意的，让模型不必为一次写入手工 mkdir。详见能力地图 delete_move_mkdir 卡。
